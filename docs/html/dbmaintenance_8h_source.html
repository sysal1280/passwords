143private:
144 void runMaintenance()
145 {
146 ScopedCursor wait(Qt::WaitCursor);
147
148 auto post = [this](const QString &text, int progress) {
149 QMetaObject::invokeMethod(this, "updateStatus",
150 Qt::QueuedConnection,
151 Q_ARG(QString, text),
152 Q_ARG(int, progress));
153 };
154
155 auto pause = []() {
156 QThread::msleep(450);
157 };
158
159 // Progress value is passed explicitly instead of reading widget state in worker thread
160 auto fail = [this, post](const QString &step, const QString &error, int progress) {
161 post(step + " failed: " + error, progress);
162
163 QMetaObject::invokeMethod(this,
164 [this, step, error]() {
165 QMessageBox::critical(this, "Maintenance Error",
166 step + " failed:\n" + error);
167 m_startBtn->setEnabled(true);
168 },
169 Qt::QueuedConnection);
170 };
171
172 QString err;
173
174 post("Running integrity check...", 1);
175 if (!runQuery("PRAGMA integrity_check;", err)) {
176 fail("Integrity check", err, 1);
177 return;
178 }
179 pause();
180
181 post("Running quick check...", 2);
182 if (!runQuery("PRAGMA quick_check;", err)) {
183 fail("Quick check", err, 2);
184 return;
185 }
186 pause();
187
188 post("Analyzing database...", 3);
189 if (!runQuery("ANALYZE;", err)) {
190 fail("Analyze", err, 3);
191 return;
192 }
193 pause();
194
195 post("Reindexing...", 4);
196 if (!runQuery("REINDEX;", err)) {
197 fail("Reindex", err, 4);
198 return;
199 }
200 pause();
201
202 post("Checking WAL mode...", 5);
203 const QString mode = getSingleValue("PRAGMA journal_mode;", err);
204 if (mode.isEmpty() && !err.isEmpty()) {
205 fail("Check WAL mode", err, 5);
206 return;
207 }
208
209 if (mode.toLower() == "wal") {
210 if (!runQuery("PRAGMA wal_checkpoint(FULL);", err)) {
211 fail("WAL checkpoint", err, 5);
212 return;
213 }
214 }
215 pause();
216
217 post("Vacuuming...", 6);
218 if (!runQuery("VACUUM;", err)) {
219 fail("Vacuum", err, 6);
220 return;
221 }
222 pause();
223 }
224
225 bool runQuery(const QString &sql, QString &errorOut)
226 {
227 QSqlQuery q(m_db);
228 if (!q.exec(sql)) {
229 errorOut = q.lastError().text();
230 return false;
231 }
232 return true;
233 }
234
235 QString getSingleValue(const QString &sql, QString &errorOut)
236 {
237 QSqlQuery q(m_db);
238 if (!q.exec(sql)) {
239 errorOut = q.lastError().text();
240 return QString();
241 }
242 if (q.next())
243 return q.value(0).toString();
244 return QString();
245 }
